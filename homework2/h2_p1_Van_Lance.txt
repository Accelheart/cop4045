3. def line_number(in_file: str, out_file: str) -> None:
4.     """
5.     The line_numer function opens the input file then goes line by line when it then will iterate each line and record 
6.     it to the output file while also keeping track of the number of the line. 
7.     """
8.     try:
9.         with open(in_file, "r") as file_in:
10.             in_collection = file_in.readlines()
11.             with open(out_file, "w") as file_out:
12.                 for line_num, text_line in enumerate(in_collection, start=1):
13.                     file_out.write(str(line_num) + ". " + text_line)
14.     except:
15.         print("No file was found")
16.         raise

18. def parse_functions(file_parse: str) -> tuple:
19.     """
20.     the parse_functions function opens the python file then reads its contents then find every function definition using the in built python
21.     module ast which is used to turn all the code into one line in which I read the line and extract information about each function
22.     using ast.FunctionDef. The function also includes line number, the function name, and the arguments. It will then remove comments and empty lines and 
23.     then sort the functions which will return the tuple at the end.
24. 
25.     """
26.     try:
27.         with open(file_parse, "r") as file:
28.             parse_collection = file.read()
29.             parse_tree = ast.parse(parse_collection)
30.             functions = []
31.             for node in ast.walk(parse_tree):
32.                 if isinstance(node, ast.FunctionDef):
33.                     line_num = node.lineno
34.                     function_name = node.name
35.                     arguments = ", ".join(arg.arg for arg in node.args.args)
36.                     function_code = ast.get_source_segment(parse_collection, node)
37.                     code_lines = function_code.splitlines()
38.                     clean_lines = []
39.                     for line in code_lines:
40.                         if "#" in line:
41.                             line = line.split("#")[0]
42.                         stripped = line.strip()
43.                         if stripped and not stripped.startswith("#"):
44.                             clean_lines.append(line)
45.                     function_code = "\n".join(clean_lines) + "\n"
46.                     functions.append((line_num, function_name, arguments, function_code))
47.             functions.sort(key=lambda x: x[1])
48.             return tuple(functions)
49.     except:
50.         print("No file was found")
51.         raise
